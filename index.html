선생님 연동 설정 (Google Sheet)

안내사항

본 웹앱은 기본적으로 브라우저(Local)에 안전하게 저장되며, 아래에 Google Apps Script 배포 URL을 입력하시면 학급 스프레드시트와 실시간 양방향 동기화됩니다.

// 구글 스프레드시트 > 확장 프로그램 > Apps Script에 붙여넣고 [웹앱] 배포 (액세스 권한: 모든 사용자)

function doGet() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  const rows = data.slice(1);
  const result = rows.map(r => ({
    id: r[0],
    nickname: r[1],
    date: r[2],
    time: r[3],
    resolution: r[4],
    photo: r[5],
    cheers: Number(r[6] || 0)
  }));
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "닉네임", "날짜", "시각", "결심", "사진", "응원수"]);
  }
  const body = JSON.parse(e.postData.contents);
  sheet.appendRow([body.id, body.nickname, body.date, body.time, body.resolution, body.photo, body.cheers || 0]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
